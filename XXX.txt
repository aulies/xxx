from telethon.tl.types import InputMessagesFilterVideo
from telethon.tl.functions.channels import GetParticipantRequest
from telethon.errors import UserNotParticipantError, ChannelInvalidError, ChannelPrivateError
import random
from .. import loader, utils

@loader.tds
class XXX(loader.Module):
    """извращенец"""

    strings = {
        "name": "XXX",
        "choosing_video": "<emoji document_id=5328311576736833844>🔴</emoji> Подбираем видео...",
        "no_videos_found": "<b>Не удалось найти видео в выбранном канале.</b>",
        "no_channel": "<b>Нет канала в конфигурации.</b>",
        "not_subscribed": "а где подписка?\n{}",
    }

    version = (1, 1, 1)

    def __init__(self):
        self.config = loader.ModuleConfig(
            loader.ConfigValue(
                "custom_channel",
                "https://t.me/+EvzOekGmcKdlNWU6",
                doc=lambda: "Введите сюда юзернейм канала или ссылку на канал.",
            )
        )

    @loader.command()
    async def x(self, message):
        """Кидает в чат пошлости 😇"""
        channel = self.config["custom_channel"]

        if not channel:
            await utils.answer(message, self.strings["no_channel"])
            return

        try:
            try:
                await self.client(GetParticipantRequest(channel, message.sender_id))
            except (UserNotParticipantError, ChannelInvalidError, ChannelPrivateError):
                await utils.answer(message, self.strings["not_subscribed"].format(channel))
                return

            choosing_msg = await utils.answer(message, self.strings["choosing_video"])

            videos = [
                msg async for msg in self.client.iter_messages(
                    channel,
                    limit=3000,
                    filter=InputMessagesFilterVideo,
                )
            ]

            if not videos:
                await utils.answer(message, self.strings["no_videos_found"])
                return

            video = random.choice(videos)

            await self.client.send_file(
                message.chat_id,
                video,
                caption=None,
                silent=True
            )

            await choosing_msg.delete()

        except Exception:
            await utils.answer(message, self.strings["not_subscribed"].format(channel))
